cmake_minimum_required(VERSION 3.24)
project(StrideImGuiNative LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The current editor native text build supports Windows x64 only.")
endif()
if(NOT EXISTS "${STRIDE_CIMGUI_SOURCE}/stride-abi-approved.json")
  message(FATAL_ERROR "Run Prepare-NativeImGui.ps1: exact pinned-header ABI approval is required.")
endif()
file(READ "${STRIDE_CIMGUI_SOURCE}/stride-abi-approved.json" ABI_CERTIFICATE)
string(JSON EXPECTED_HEADER_HASH GET "${ABI_CERTIFICATE}" GeneratedHeaderSha256)
file(SHA256 "${STRIDE_CIMGUI_SOURCE}/cimgui.h" ACTUAL_HEADER_HASH)
string(TOUPPER "${ACTUAL_HEADER_HASH}" ACTUAL_HEADER_HASH)
if(NOT ACTUAL_HEADER_HASH STREQUAL EXPECTED_HEADER_HASH)
  message(FATAL_ERROR "Generated cimgui header changed after ABI approval.")
endif()
string(JSON INPUT_COUNT LENGTH "${ABI_CERTIFICATE}" Inputs)
math(EXPR LAST_INPUT "${INPUT_COUNT} - 1")
foreach(INDEX RANGE ${LAST_INPUT})
  string(JSON INPUT_PATH MEMBER "${ABI_CERTIFICATE}" Inputs ${INDEX})
  string(JSON EXPECTED_HASH GET "${ABI_CERTIFICATE}" Inputs "${INPUT_PATH}")
  file(SHA256 "${STRIDE_CIMGUI_SOURCE}/${INPUT_PATH}" ACTUAL_HASH)
  string(TOUPPER "${ACTUAL_HASH}" ACTUAL_HASH)
  if(NOT ACTUAL_HASH STREQUAL EXPECTED_HASH)
    message(FATAL_ERROR "Native source ${INPUT_PATH} changed after pinned-source approval.")
  endif()
endforeach()
set(IMGUI_WCHAR32 ON CACHE BOOL "" FORCE)
set(IMGUI_FREETYPE ON CACHE BOOL "" FORCE)
set(CIMGUI_VARGS0 ON CACHE BOOL "" FORCE)
set(CIMGUI_TEST OFF CACHE BOOL "" FORCE)
add_subdirectory("${STRIDE_CIMGUI_SOURCE}" cimgui-build)
target_sources(cimgui PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/StrideImGuiTextHooks.cpp")
target_include_directories(cimgui PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
set_property(TARGET cimgui PROPERTY CXX_STANDARD 11)
